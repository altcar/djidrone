Hugging Face dataset (gated)

  • uploaded/ — 119 original DJI 4K HEVC videos (~220 GB).
  • PANORAMA/ — 76 JPGs in 4 sets: 001_0002, 001_0005, 001_0081, 001_0089.

GitHub site

https://altcar.github.io/djidrone · repo altcar/djidrone

  • Gallery of all videos with 4K / HEVC badges; streams the original HEVC files through the Cloudflare proxy — no transcoding.
  • Password gate: rich (PBKDF2-SHA256, 310,000 iterations, random salt). Only the salt + hash are stored in auth.json; the plaintext is never in the repo.
  • Access lasts for the tab session, or 12 hours on this device with “Remember this device”. The 🔒 Lock button re-locks immediately.
  • Panorama puzzle arrangements are stored in the viewer's browser (localStorage) — there is no gist or server.
  • tools/set_password.py regenerates auth.json with a new salt and hash; commit and push that file to change the password. Source is reproduced below.

Cloudflare Worker (no local bridge needed)

https://djidrone-video.hheylau.workers.dev

  • Holds the HF token as a server-side secret (never exposed to browsers), enforces the gallery password before streaming, and forwards HTTP Range requests so seeking works. The dataset stays gated.
  • Workers cap PBKDF2 at 100,000 iterations, so the Worker uses a separate 100k-iteration hash of rich stored as AUTH_JSON_TEXT. If you change the gallery password, you must also update this Worker secret.
  • Worker source: worker/worker.js (reproduced below). Deploy with npx wrangler deploy from the worker/ directory.

tools/set_password.py

#!/usr/bin/env python3
"""Set or change the gallery password (password-only gate, no username).

Usage:
  python3 tools/set_password.py                 # prompts twice (hidden)
  GATE_PASSWORD='...' python3 tools/set_password.py   # non-interactive

Only a random salt + PBKDF2-SHA256 hash is written to auth.json. The password
itself is never stored.
"""
import getpass
import hashlib
import json
import os
import secrets
from pathlib import Path

ITERATIONS = 310_000
OUT = Path(__file__).resolve().parent.parent / "auth.json"

pw = os.environ.get("GATE_PASSWORD")
if pw is None:
    pw = getpass.getpass("New gallery password: ")
    again = getpass.getpass("Repeat password: ")
    if pw != again:
        raise SystemExit("Passwords do not match.")
if not pw:
    raise SystemExit("Password must not be empty.")

salt = secrets.token_bytes(16)
digest = hashlib.pbkdf2_hmac("sha256", pw.encode("utf-8"), salt, ITERATIONS, dklen=32)
OUT.write_text(json.dumps({
    "salt": salt.hex(),
    "hash": digest.hex(),
    "iterations": ITERATIONS,
    "algorithm": "PBKDF2-SHA256",
}, indent=2) + "\\n")
print(f"Updated {OUT} (salt + hash only; password not stored).")

worker/worker.js

/* Private video proxy for the gated djidrone HF dataset.
   - HF token is a Worker secret: never sent to the browser.
   - Requests are gated by the gallery password (PBKDF2 hash verified with
     Web Crypto-equivalent crypto.subtle in Workers).
   - Forwards Range requests so <video> can seek without downloading 4K files.
   - Only paths under uploaded/ and playback/ are allowed. */

const ALLOWED = /^(uploaded|playback)\/[^?]+\.(mp4|mov|webm)$/i;

function json(body, status = 200, extra = {}) {
  return new Response(JSON.stringify(body), {
    status,
    headers: { "content-type": "application/json", "access-control-allow-origin": "*", ...extra },
  });
}

async function verifyPassword(request, env, url) {
  const auth = await env.AUTH.get("auth.json");
  if (!auth) return { ok: false, status: 500, error: "auth config missing" };

  let password = "";
  const header = request.headers.get("Authorization") || "";
  if (header.startsWith("Bearer ")) {
    try { password = atob(header.slice(7)); } catch (e) { /* fall through to query */ }
  }
  if (!password) {
    // <video> elements cannot send custom headers; the unlocked gallery passes
    // the password as a query parameter to this same-origin-protected proxy.
    password = url.searchParams.get("auth") || "";
  }
  if (!password) return { ok: false, status: 401, error: "password required" };

  const cfg = JSON.parse(auth);
  const enc = new TextEncoder();
  const key = await crypto.subtle.importKey("raw", enc.encode(password), "PBKDF2", false, ["deriveBits"]);
  const bits = await crypto.subtle.deriveBits(
    { name: "PBKDF2", salt: hexToBytes(cfg.salt), iterations: cfg.iterations, hash: "SHA-256" },
    key, 256
  );
  const hash = bytesToHex(new Uint8Array(bits)).toLowerCase();
  if (hash !== String(cfg.hash).toLowerCase()) return { ok: false, status: 403, error: "wrong password" };
  return { ok: true };
}

function hexToBytes(hex) {
  const out = new Uint8Array(hex.length / 2);
  for (let i = 0; i < out.length; i++) out[i] = parseInt(hex.slice(i * 2, i * 2 + 2), 16);
  return out;
}
function bytesToHex(b) {
  return Array.prototype.map.call(b, (x) => ("0" + x.toString(16)).slice(-2)).join("");
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    if (url.pathname === "/health") return json({ ok: true, repo: env.HF_REPO });

    // CORS preflight for the browser player.
    if (request.method === "OPTIONS") {
      return new Response(null, {
        status: 204,
        headers: {
          "access-control-allow-origin": "*",
          "access-control-allow-headers": "Authorization, Range",
          "access-control-allow-methods": "GET, HEAD, OPTIONS",
          "access-control-expose-headers": "Content-Length, Content-Range, Accept-Ranges",
        },
      });
    }

    if (request.method !== "GET" && request.method !== "HEAD") return json({ error: "method not allowed" }, 405);

    const auth = await verifyPassword(request, env, url);
    if (!auth.ok) return json({ error: auth.error }, auth.status);

    const rel = decodeURIComponent(url.pathname.replace(/^\/+/, ""));
    if (!ALLOWED.test(rel)) return json({ error: "invalid path" }, 400);

    const target = `https://huggingface.co/datasets/${env.HF_REPO}/resolve/main/${rel}`;
    const headers = { Authorization: `Bearer ${env.HF_TOKEN}` };
    const range = request.headers.get("Range");
    if (range && /^bytes=\d*-\d*$/.test(range)) headers.Range = range;

    const upstream = await fetch(target, { headers, method: "GET" });
    if (!upstream.ok && upstream.status !== 206) {
      return json({ error: "HF returned " + upstream.status }, upstream.status);
    }

    const outHeaders = new Headers({
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Expose-Headers": "Content-Length, Content-Range, Accept-Ranges",
      "Cache-Control": "private, max-age=3600",
    });
    for (const k of ["content-type", "content-length", "content-range", "accept-ranges", "etag"]) {
      const v = upstream.headers.get(k);
      if (v) outHeaders.set(k, v);
    }
    return new Response(request.method === "HEAD" ? null : upstream.body, {
      status: upstream.status,
      headers: outHeaders,
    });
  },
};
